#### Docker Compose配置

```yaml
version: '3.8'

services:
  # 前端静态资源服务
  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
      args:
        - NODE_ENV=production
    volumes:
      - ./frontend/dist:/usr/share/nginx/html
    restart: unless-stopped
    networks:
      - app-network

  # 后端API服务
  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    environment:
      - DB_HOST=mysql
      - DB_PORT=3306
      - DB_NAME=${DB_NAME}
      - DB_USER=${DB_USER}
      - DB_PASSWORD=${DB_PASSWORD}
      - REDIS_HOST=redis
      - REDIS_PORT=6379
    depends_on:
      mysql:
        condition: service_healthy
      redis:
        condition: service_started
    restart: unless-stopped
    networks:
      - app-network

  # MySQL数据库
  mysql:
    image: mysql:8.0
    command: --default-authentication-plugin=mysql_native_password
    environment:
      - MYSQL_ROOT_PASSWORD=${DB_ROOT_PASSWORD}
      - MYSQL_DATABASE=${DB_NAME}
      - MYSQL_USER=${DB_USER}
      - MYSQL_PASSWORD=${DB_PASSWORD}
    volumes:
      - mysql_data:/var/lib/mysql
      - ./sql/init.sql:/docker-entrypoint-initdb.d/init.sql:ro
    ports:
      - "3306:3306"
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      timeout: 20s
      retries: 10
    restart: unless-stopped
    networks:
      - app-network

  # Redis缓存
  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes
    volumes:
      - redis_data:/data
    ports:
      - "6379:6379"
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 5
    restart: unless-stopped
    networks:
      - app-network

  # Nginx反向代理
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./nginx/ssl:/etc/nginx/ssl:ro
      - ./frontend/dist:/usr/share/nginx/html:ro
    depends_on:
      - backend
      - frontend
    restart: unless-stopped
    networks:
      - app-network

volumes:
  mysql_data:
  redis_data:

networks:
  app-network:
    driver: bridge
```

### 2.2 后端Dockerfile

```dockerfile
# 多阶段构建

# 构建阶段
FROM golang:1.21-alpine AS builder

# 安装编译依赖
RUN apk add --no-cache git make

# 设置工作目录
WORKDIR /build

# 复制依赖文件
COPY go.mod go.sum ./
RUN go mod download

# 复制源代码
COPY . .

# 编译
RUN CGO_ENABLED=0 GOOS=linux go build -o server .

# 运行阶段
FROM alpine:latest

# 安装CA证书（用于HTTPS请求）
RUN apk --no-cache add ca-certificates tzdata

# 设置时区
ENV TZ=Asia/Shanghai

# 创建非root用户
RUN addgroup -g 1000 appuser && \
    adduser -D -u 1000 -G appuser appuser

WORKDIR /app

# 从构建阶段复制二进制文件
COPY --from=builder /build/server .

# 更改文件所有者
RUN chown -R appuser:appuser /app

# 切换用户
USER appuser

# 暴露端口
EXPOSE 8080

# 健康检查
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
  CMD wget --no-verbose --tries=1 --spider http://localhost:8080/health || exit 1

# 启动服务
CMD ["./server"]
```

## 3. Nginx反向代理

### 3.1 负载均衡

```nginx
# upstream配置
upstream backend_servers {
    # 负载均衡策略：least_conn（最少连接）
    least_conn;

    # 后端服务器列表（可扩展为多实例）
    server backend:8080 max_fails=3 fail_timeout=30s;
    # server backend2:8080 max_fails=3 fail_timeout=30s;
}

upstream frontend_servers {
    # 前端服务器
    server frontend:80;
}

# keepalive连接池
upstream backend_keepalive {
    server backend:8080;
    keepalive 32;
}
```

### 3.2 SSL配置

#### 生成自签名证书（开发环境）

```bash
# 生成私钥
openssl genrsa -out nginx/ssl/private.key 2048

# 生成证书
openssl req -new -key nginx/ssl/private.key \
    -out nginx/ssl/certificate.csr \
    -subj "/C=CN/ST=Beijing/L=Beijing/O=Elderly Care/CN=localhost"

# 生成自签名证书
openssl x509 -req -days 365 -in nginx/ssl/certificate.csr \
    -signkey nginx/ssl/private.key \
    -out nginx/ssl/certificate.crt
```

#### Let's Encrypt证书（生产环境）

```bash
# 安装certbot
sudo apt install certbot python3-certbot-nginx

# 获取证书
```

### 3.3 完整Nginx配置

```nginx
user nginx;
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 2048;
    use epoll;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    # 日志格式
    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;

    # 性能优化
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;
    client_max_body_size 20M;

    # Gzip压缩
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_proxied any;
    gzip_types text/plain text/css text/xml text/javascript
               application/json application/javascript application/xml+rss
               application/rss+xml font/truetype font/opentype
               application/vnd.ms-fontobject image/svg+xml;

    # 限流配置
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=10r/s;
    limit_req_zone $binary_remote_addr zone=general_limit:10m rate=30r/s;
    limit_conn_zone $binary_remote_addr zone=conn_limit:10m rate=10;

    # 前端服务器
    upstream frontend {
        server frontend:80;
    }

    # 后端API服务器
    upstream backend {
        server backend:8080;
    }

    # HTTP重定向到HTTPS
    server {
        listen 80;
        server_name your-domain.com;
        return 301 https://$server_name$request_uri;
    }

    # HTTPS配置
    server {
        listen 443 ssl http2;
        server_name your-domain.com;

        # SSL证书
        ssl_certificate /etc/nginx/ssl/certificate.crt;
        ssl_certificate_key /etc/nginx/ssl/private.key;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;
        ssl_prefer_server_ciphers on;
        ssl_session_cache shared:SSL:10m;
        ssl_session_timeout 10m;

        # 安全头
        add_header X-Frame-Options DENY;
        add_header X-Content-Type-Options nosniff;
        add_header X-XSS-Protection "1; mode=block";
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

        # 前端静态资源
        location / {
            limit_req zone=general_limit burst=50 nodelay;
            limit_conn conn_limit 10;

            proxy_pass http://frontend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # 缓存配置
            expires 7d;
            add_header Cache-Control "public, immutable";
        }

        # API接口
        location /api {
            limit_req zone=api_limit burst=20 nodelay;
            limit_conn conn_limit 5;

            proxy_pass http://backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # 超时配置
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;

            # 禁用缓存
            add_header Cache-Control "no-cache, no-store, must-revalidate";
        }

        # WebSocket支持
        location /ws {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_read_timeout 86400;
        }

        # 健康检查
        location /health {
            access_log off;
            return 200 "healthy\n";
            add_header Content-Type text/plain;
        }
    }
}
```

### 5.3 监控脚本

```bash
#!/bin/bash
# monitor.sh - 系统监控脚本

# 检查服务状态
check_services() {
    echo "=== 服务状态检查 ==="
    docker-compose ps
}

# 检查资源使用
check_resources() {
    echo "=== 资源使用情况 ==="
    echo "CPU使用率："
    top -bn1 | grep "Cpu(s)"

    echo "内存使用："
    free -h

    echo "磁盘使用："
    df -h
}

# 检查Nginx状态
check_nginx() {
    echo "=== Nginx状态 ==="
    curl -s http://localhost/health
}

# 检查API健康
check_api() {
    echo "=== API健康检查 ==="
    curl -s http://localhost:8080/health
}

# 主函数
main() {
    case "$1" in
        all)
            check_services
            check_resources
            check_nginx
            check_api
            ;;
        services)
            check_services
            ;;
        resources)
            check_resources
            ;;
        nginx)
            check_nginx
            ;;
        api)
            check_api
            ;;
        *)
            echo "用法: $0 {all|services|resources|nginx|api}"
            exit 1
            ;;
    esac
}

main "$@"
```

### 5.4 日志管理

```bash
#!/bin/bash
# log-rotate.sh - 日志轮转脚本

LOG_DIR="/var/log/nginx"
BACKUP_DIR="/var/log/nginx/archive"

# 创建备份目录
mkdir -p $BACKUP_DIR

# 轮转Nginx日志
rotate_nginx_logs() {
    for logfile in access.log error.log; do
        if [ -f "$LOG_DIR/$logfile" ]; then
            # 压缩并移动旧日志
            timestamp=$(date +%Y%m%d_%H%M%S)
            gzip -c "$LOG_DIR/$logfile" > "$BACKUP_DIR/${logfile}.${timestamp}.gz"

            # 清空原日志文件
            > "$LOG_DIR/$logfile"

            echo "已轮转日志: $logfile"
        fi
    done

    # 重启Nginx以重新打开日志文件
    docker-compose exec nginx nginx -s reload
}

# 删除30天前的备份日志
cleanup_old_logs() {
    find $BACKUP_DIR -name "*.gz" -mtime +30 -delete
    echo "已清理30天前的日志"
}

# 主函数
main() {
    rotate_nginx_logs
    cleanup_old_logs
}

main
```

## 6. 备份与恢复

### 6.1 数据备份策略

```bash
#!/bin/bash
# backup.sh - 数据库备份脚本

BACKUP_DIR="/backup/mysql"
DATE=$(date +%Y%m%d_%H%M%S)
DB_NAME="elderly_care"

# 创建备份目录
mkdir -p $BACKUP_DIR

# 备份数据库
docker-compose exec -T mysql mysqldump -u root -p${DB_ROOT_PASSWORD} $DB_NAME \
    | gzip > $BACKUP_DIR/${DB_NAME}_${DATE}.sql.gz

# 删除7天前的备份
find $BACKUP_DIR -name "*.sql.gz" -mtime +7 -delete

echo "数据库备份完成: ${DB_NAME}_${DATE}.sql.gz"
```

### 6.2 定时备份（Cron）

```bash

```

### 6.3 数据恢复

```bash
#!/bin/bash
# restore.sh - 数据库恢复脚本

if [ -z "$1" ]; then
    echo "用法: $0 <backup_file>"
    exit 1
fi

BACKUP_FILE=$1

# 解压并恢复数据库
gunzip -c $BACKUP_FILE | docker-compose exec -T mysql mysql -u root -p${DB_ROOT_PASSWORD} elderly_care

echo "数据库恢复完成"
```
