## 1. 设计原则

### 1.1 老年人友好设计

| 设计要素 | 具体要求 | 实现方式 |
|---------|---------|---------|
| **大字体** | 基础字号18px，标题24px+ | CSS `font-size: 18px` |
| **高对比度** | WCAG AA级对比度（4.5:1） | 深色文字+浅色背景 |
| **简化流程** | 单页提交，最多3步 | 表单分页引导 |
| **清晰反馈** | 操作后1秒内显示反馈 | Toast提示+状态图标 |
| **大按钮** | 最小点击区域48x48px | CSS `min-height: 48px` |
| **语音辅助** | 支持语音输入 | Web Speech API |
| **色彩明确** | 使用语义化颜色 | 成功=绿，警告=橙，错误=红 |

### 1.2 PWA能力设计

| 能力 | 价值 | 实现方式 |
|------|------|---------|
| **离线访问** | 弱网环境可用 | Service Worker预缓存 |
| **安装到桌面** | 减少操作步骤 | Web App Manifest |
| **消息推送** | 及时通知 | Push API + Notification |
| **后台同步** | 数据不丢失 | Background Sync API |
| **添加到主屏幕** | 快速访问 | Manifest icons配置 |

```vue
import { useRouter } from 'vue-router'
import { createRequest } from '@/api/requests'

const router = useRouter()
const currentStep = ref(1)
const showConfirm = ref(false)

const formData = reactive({
  service_type: '',
  title: '',
  description: '',
  contact_name: '',
  contact_phone: '',
  address: '',
  location: null,
  service_time: '',
  special_requirements: '',

const handleServiceTypeSelect = (type: string) => {
  formData.service_type = type
  // 自动填充标题

    meal: '助餐服务',
    cleaning: '保洁服务',
    medical: '医疗护理',
    accompany: '陪伴服务',
    other: '其他服务'
  }
  formData.title = titles[type] || '服务需求'
}

const nextStep = async () => {
  // 验证当前步骤
  const valid = await validateStep(currentStep.value)
  if (valid) {
    if (currentStep.value < 3) {
      currentStep.value++
    } else {
      showConfirm.value = true
    }
  }

const prevStep = () => {
  if (currentStep.value > 1) {

const handleSubmit = async () => {

    const result = await createRequest(formData)
    router.push(`/success/${result.id}`)
  } catch (error) {
```

### 4.3 离线能力（IndexedDB）

#### 离线存储封装

```typescript
// src/utils/indexedDB.ts
import { openDB, DBSchema, IDBPDatabase } from 'idb'

interface RequestDB extends DBSchema {
  requests: {
    key: string
    value: {
      id: string
      data: any
      timestamp: number
      synced: boolean
    }
    indexes: {
      'by-synced': boolean
    }
  }
}

class OfflineStorage {
  private db: IDBPDatabase<RequestDB> | null = null

  async init() {
    this.db = await openDB<RequestDB>('elderly-care-requests', 1, {
      upgrade(db) {
        const store = db.createObjectStore('requests', {
          keyPath: 'id'
        })
        store.createIndex('by-synced', 'synced')
      }
    })
  }

  async saveRequest(id: string, data: any) {
    if (!this.db) await this.init()
    return this.db!.put('requests', {
      id,
      data,
      timestamp: Date.now(),
      synced: false
    })
  }

  async getUnsyncedRequests() {
    if (!this.db) await this.init()
    return this.db!.getAllFromIndex('requests', 'by-synced', false)
  }

  async markAsSynced(id: string) {
    if (!this.db) await this.init()
    const item = await this.db!.get('requests', id)
    if (item) {
      item.synced = true
      return this.db!.put('requests', item)
    }
  }

  async deleteRequest(id: string) {
    if (!this.db) await this.init()
    return this.db!.delete('requests', id)
  }
}

export const offlineStorage = new OfflineStorage()
```

#### 后台同步

```typescript
// src/utils/sw-events.ts
import { offlineStorage } from './indexedDB'

export function setupEventListeners(sw: ServiceWorkerGlobalScope) {
  sw.addEventListener('sync', (event) => {
    if (event.tag === 'sync-requests') {
      event.waitUntil(syncRequests(sw))
    }
  })
}

async function syncRequests(sw: ServiceWorkerGlobalScope) {
  const unsynced = await offlineStorage.getUnsyncedRequests()

  for (const item of unsynced) {
    try {
      const response = await fetch('https://api.elderly-care.com/api/requests', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(item.data)
      })

      if (response.ok) {
        await offlineStorage.markAsSynced(item.id)
      }
    } catch (error) {
      console.error('[SW] 同步失败:', error)
    }
  }
}
```

### 4.4 语音输入

```vue
<template>
  <div class="voice-input">
    <textarea
      v-model="text"
      :placeholder="placeholder"
      :disabled="isListening"
    />
    <button
      :class="['voice-button', { listening: isListening }]"
      @click="toggleVoice"
    >
      <span v-if="isListening">🎤 停止</span>
      <span v-else>🎤 语音输入</span>
    </button>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'

const props = defineProps<{
  modelValue: string
  placeholder?: string
}>()

const emit = defineEmits<{
  'update:modelValue': [value: string]
}>()

const text = ref(props.modelValue)
const isListening = ref(false)
let recognition: any = null

const initRecognition = () => {
  if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
    alert('您的浏览器不支持语音输入')
    return null
  }

  const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
  recognition = new SpeechRecognition()
  recognition.lang = 'zh-CN'
  recognition.continuous = false
  recognition.interimResults = false

  recognition.onresult = (event: any) => {
    const transcript = event.results[0][0].transcript
    text.value = text.value + transcript
    emit('update:modelValue', text.value)
  }

  recognition.onerror = (event: any) => {
    console.error('语音识别错误:', event.error)
    isListening.value = false
  }

  recognition.onend = () => {
    isListening.value = false
  }

  return recognition
}

const toggleVoice = () => {
  if (isListening.value) {
    recognition?.stop()
    isListening.value = false
  } else {
    if (!recognition) {
      recognition = initRecognition()
    }
    recognition?.start()
    isListening.value = true
  }
}
</script>
```
