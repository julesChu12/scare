## 0. 当前实现状态（与代码一致）

**代码位置**：`frontend/management-portal/`

**技术栈（已实现）**：Vue 3 + TypeScript + Vite + Element Plus + Pinia

**已实现路由**（`frontend/management-portal/src/router/index.ts`）：

```
/login
/           -> /tasks/pool
/tasks/pool  # 任务池
/tasks/my    # 我的任务
/tasks/:id   # 任务详情
```

**已实现功能**：

- 登录与基础布局（BasicLayout）
- 任务池、我的任务、任务详情
- 任务认领与完成接口调用
- 角色路由守卫（基于 meta.roles）

**未实现/规划中**：

- 资源级权限校验已通过自定义 RBAC 三表模型实现（permissions/roles/role_permissions）
- 工作台/统计/站点/围栏/用户管理
- 地图围栏编辑（仅有 MapViewer 组件示例）
- ECharts 数据可视化页面

#### 2.3.2 地图编辑组件

```vue
<template>
  <div class="zone-map-editor">
    <!-- 地图容器 -->
    <div ref="mapContainer" class="map-container"></div>

    <!-- 工具栏 -->
    <div class="map-toolbar">
      <button @click="startDraw">开始绘制</button>
      <button @click="saveZone" :disabled="!hasZone">保存</button>
      <button @click="clearZone">清除</button>
    </div>

    <!-- 围栏信息 -->
    <div v-if="currentZone" class="zone-info">
      <div>顶点数：{{ currentZone.path.length }}</div>
      <div>估算面积：{{ calculateArea(currentZone.path) }}</div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onUnmounted } from 'vue'
import AMapLoader from '@amap/amap-jsapi-loader'

const mapContainer = ref<HTMLElement>()
const map = ref<any>()
const drawTool = ref<any>()
const currentZone = ref<any>()
const hasZone = ref(false)

onMounted(async () => {
  await AMapLoader.load({
    key: import.meta.env.VITE_AMAP_KEY,
    version: '2.0',
    plugins: ['AMap.PolygonEditor']
  })

  map.value = new AMap.Map(mapContainer.value, {
    zoom: 15,
    center: [116.3667, 40.0833]
  })

  drawTool.value = new AMap.PolygonEditor(map.value)
})

const startDraw = () => {
  drawTool.value.open()
  drawTool.value.addListener('add', handlePolygonAdd)
}

const handlePolygonAdd = (e: any) => {
  const polygon = e.target
  const path = polygon.getPath()
  currentZone.value = {
    path: path.map((p: any) => ({
      lng: p.lng,
      lat: p.lat
    }))
  }
  hasZone.value = true
}

const saveZone = () => {
  emit('save', currentZone.value)
}

const clearZone = () => {
  drawTool.value.clear()
  currentZone.value = null
  hasZone.value = false
}

const calculateArea = (path: any[]) => {
  // 使用高德地图工具计算面积
  const polygon = new AMap.Polygon(path)
  return AMap.GeometryUtil.polygonArea(polygon).toFixed(2) + ' 平方米'
}
</script>
```

<template>
  <div class="map-wrapper">

    <!-- 地图控件 -->

      <button @click="zoomIn" class="control-btn">+</button>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import AMapLoader from '@amap/amap-jsapi-loader'

const props = defineProps<{
  stations: Station[]
  zones: Zone[]
}>()

const mapRef = ref<HTMLElement>()
const map = ref<any>()
const markers = ref<any[]>([])
const polygons = ref<any[]>([])
const selectedMarker = ref<Station | null>(null)

onMounted(async () => {
  await initMap()
  renderStations()
  renderZones()
})

const initMap = async () => {
  await AMapLoader.load({
    key: import.meta.env.VITE_AMAP_KEY,
    version: '2.0',
    plugins: ['AMap.Marker', 'AMap.Polygon', 'AMap.InfoWindow']
  })

  map.value = new AMap.Map(mapRef.value, {
    zoom: 14,
    center: [116.3667, 40.0833],
    mapStyle: 'amap://styles/normal'
  })

  // 添加工具栏
  map.value.addControl(new AMap.ToolBar({
    position: 'RT'
  }))
}

const renderStations = () => {
  props.stations.forEach(station => {
    const marker = new AMap.Marker({
      position: [station.longitude, station.latitude],
      title: station.name,
      icon: new AMap.Icon({
        image: '/station-marker.png',
        size: new AMap.Size(32, 32),
        imageSize: new AMap.Size(32, 32)
      })
    })

    marker.on('click', () => {
      selectedMarker.value = station
    })

    marker.setMap(map.value)
    markers.value.push(marker)
  })
}

const renderZones = () => {
  props.zones.forEach(zone => {
    const path = zone.coordinates.map(coord => [coord.lng, coord.lat])

    const polygon = new AMap.Polygon({
      path: path,
      strokeColor: '#1976d2',
      strokeWeight: 2,
      fillColor: 'rgba(25, 118, 210, 0.1)',
      strokeStyle: 'solid'
    })

    polygon.setMap(map.value)
    polygons.value.push(polygon)
  })
}

const zoomIn = () => map.value.zoomIn()
const zoomOut = () => map.value.zoomOut()
const resetView = () => {
  map.value.setZoomAndCenter(14, [116.3667, 40.0833])
}
const closeInfo = () => {
  selectedMarker.value = null
}
</script>

      },
      {

        name: 'stations',
        component: () => import('@/views/Stations.vue'),

      },
      {
        path: 'users',
        name: 'users',
        component: () => import('@/views/Users.vue'),
        meta: { title: '用户管理', roles: ['admin'] }
      },
      {
        path: 'statistics',
        name: 'statistics',
        component: () => import('@/views/Statistics.vue'),
        meta: { title: '统计分析', roles: ['staff', 'admin'] }
      }
    ]
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

// 路由守卫
router.beforeEach((to, from, next) => {
  const userStore = useUserStore()

  if (to.meta.requiresAuth && !userStore.isLoggedIn) {
    next({ name: 'login', query: { redirect: to.fullPath } })
    return
  }

  // 权限检查
  if (to.meta.roles) {
    if (!to.meta.roles.includes(userStore.user?.role)) {
      next({ name: 'dashboard' })
      return
    }
  }

  // 权限字符串检查
  if (to.meta.permission) {
    if (!userStore.hasPermission(to.meta.permission)) {
      next({ name: 'dashboard' })
      return
    }
  }

  next()
})

export default router

### 4.3 权限指令

// src/directives/permission.ts
import type { Directive } from 'vue'
import { useUserStore } from '@/stores/user'

export const permission: Directive = {
  mounted(el, binding) {
    const { value } = binding
    const userStore = useUserStore()

    if (value && !userStore.hasPermission(value)) {
      el.parentNode?.removeChild(el)
    }
  }
}

// 注册指令
// src/main.ts
app.directive('permission', permission)
